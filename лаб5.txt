ЧАСТЬ 2. ЗАПРОСЫ НА ВЫБОРКУ (SELECT)

Неявное соединение таблиц с фильтром данных (WHERE) и сортировкой (ORDER BY):

-- Специалист техподдержки формирует список всех заявок конкретного заказчика (ЛУКОЙЛ-Пермнефтепродукт), чтобы понять историю обращений, отсортировав их от новых к старым:
SELECT r.request AS "Номер заявки", c.name AS "Заказчик", r.request_date AS "Дата заявки", r.problem_description AS "Проблема" 
FROM `request` r, `customer` c 
WHERE r.customer = c.customer AND c.name = 'ЛУКОЙЛ-Пермнефтепродукт' 
ORDER BY r.request_date DESC;

-- Руководитель проекта смотрит заявки с высоким приоритетом, чтобы назначить их в первую очередь:
SELECT r.request AS "Номер заявки", rp.name AS "Приоритет", r.planned_date AS "Планируемая дата", r.execution_priority AS "Срочность" 
FROM `request` r, `request_priority` rp 
WHERE r.priority = rp.priority AND rp.name = 'Высокий' 
ORDER BY r.planned_date ASC;

-- Заявки по конкретному проекту:
SELECT r.request AS "Номер заявки", p.name AS "Проект", r.problem_description AS "Описание проблемы", r.request_date AS "Дата" 
FROM `request` r, `project` p 
WHERE r.project = p.project AND p.name = 'Petronics 2.0' 
ORDER BY r.request_date DESC;


Явное соединение таблиц (JOIN):

-- Формирование полной карточки заявок (проблема, категория, статус, исполнитель) для директора:
SELECT r.request AS "Номер", r.problem_description AS "Проблема", rc.name AS "Категория", rs.name AS "Статус", e.full_name AS "Исполнитель" 
FROM `request` r 
INNER JOIN `request_category` rc ON r.category = rc.category 
INNER JOIN `request_status` rs ON r.status = rs.status 
INNER JOIN `employee` e ON r.employee = e.employee;

-- Журнал трудозатрат (кто, какую услугу выполнял и сколько часов потратил):
SELECT w.fix_date AS "Дата", e.full_name AS "Сотрудник", s.name AS "Услуга", w.hours AS "Кол-во часов" 
FROM `worklog` w 
INNER JOIN `request` r ON w.request = r.request 
INNER JOIN `employee` e ON r.employee = e.employee 
INNER JOIN `service` s ON w.service = s.service;

-- Вывести полный отчет по заявке для директора:
SELECT r.request AS "Заявка", c.name AS "Заказчик", eq.name AS "Оборудование", e.full_name AS "Назначенный исполнитель" 
FROM `request` r 
INNER JOIN `customer` c ON r.customer = c.customer 
INNER JOIN `equipment` eq ON r.equipment = eq.equipment 
INNER JOIN `employee` e ON r.employee = e.employee;


Агрегирующие операции (SUM, AVG, COUNT, MAX, MIN):

-- Руководитель направления анализирует общую статистику:
SELECT COUNT(*) AS "Всего заявок в базе", 
       (SELECT AVG(price) FROM `service`) AS "Средняя стоимость услуги", 
       (SELECT MAX(price) FROM `service`) AS "Самая дорогая услуга (руб)", 
       (SELECT MIN(price) FROM `service`) AS "Самая дешевая услуга (руб)" 
FROM `request`;

-- Анализ трудозатрат: сколько часов потрачено в сумме и в среднем на одну услугу:
SELECT SUM(hours) AS "Всего потрачено часов", AVG(hours) AS "Среднее часов на 1 услугу" 
FROM `worklog`;

-- Макс. и мин. оклады:
SELECT MAX(salary) AS "Максимальный оклад", MIN(salary) AS "Минимальный оклад" 
FROM `position`;


Группировка GROUP BY и фильтрация HAVING:

-- Количество заявок в каждом статусе:
SELECT rs.name AS "Статус заявки", COUNT(r.request) AS "Количество заявок" 
FROM `request_status` rs 
LEFT JOIN `request` r ON rs.status = r.status 
GROUP BY rs.name;

-- Руководитель проекта ищет перегруженных сотрудников, кто отработал больше 2 часов:
SELECT e.full_name AS "Сотрудник", SUM(w.hours) AS "Суммарные часы" 
FROM `employee` e 
INNER JOIN `request` r ON e.employee = r.employee 
INNER JOIN `worklog` w ON r.request = w.request 
GROUP BY e.full_name 
HAVING SUM(w.hours) > 2;

-- Выявление самых частых проблем:
SELECT rc.name AS "Категория", COUNT(r.request) AS "Кол-во заявок" 
FROM `request_category` rc 
INNER JOIN `request` r ON rc.category = r.category 
GROUP BY rc.name 
HAVING COUNT(r.request) > 1;

-- Выявление часто обращающихся клиентов:
SELECT c.name AS "Заказчик", COUNT(r.request) AS "Всего заявок" 
FROM `customer` c 
INNER JOIN `request` r ON c.customer = r.customer 
GROUP BY c.name 
HAVING COUNT(r.request) > 1;


Вложенные подзапросы (запрос в запросе), в т.ч. использовать операторы IN, ALL:

-- IN: Найти всех заказчиков, у которых есть хотя бы одна заявка в статусе «в работе» (3):
SELECT name AS "Заказчик", contact_person AS "Контактное лицо" 
FROM `customer` 
WHERE customer IN (
    SELECT customer FROM `request` WHERE status = 3
);

-- IN: Найти ФИО сотрудников, которые когда-либо выполняли «диагностику» (3):
SELECT full_name AS "Сотрудник", position AS "Должность" 
FROM `employee` 
WHERE employee IN (
    SELECT r.employee FROM `request` r 
    INNER JOIN `worklog` w ON r.request = w.request 
    WHERE w.service = 3
);

-- ALL: Найти заявку, на которую было потрачено больше всего часов:
SELECT request AS "Номер заявки", hours AS "Количество часов" 
FROM `worklog` 
WHERE hours >= ALL (SELECT hours FROM `worklog`);


Использование «функций управления потоком» (IF, CASE) в соответствии с бизнес-логикой задач:

-- CASE: Визуализация срочности заявки:
SELECT request AS "Номер заявки", problem_description AS "Проблема", 
       CASE 
           WHEN priority = 4 THEN 'Горит!' 
           WHEN priority = 1 THEN 'Горит!' 
           WHEN priority = 3 THEN 'В порядке очереди' 
           WHEN priority = 2 THEN 'В порядке очереди' 
           ELSE 'Не указано' 
       END AS "Срочность" 
FROM `request`;

-- CASE: Статус активности по цвету:
SELECT r.problem_description AS "Проблема", 
       CASE 
           WHEN rs.color = 'Зеленый' THEN 'Решена' 
           WHEN rs.color = 'Синий' THEN 'Поступила' 
           WHEN rs.color = 'Желтый' THEN 'Решается' 
           WHEN rs.color = 'Оранжевый' THEN 'Проверяется' 
           ELSE 'Не указано' 
       END AS "Активность" 
FROM `request` r 
INNER JOIN `request_status` rs ON r.status = rs.status;

-- IF: Разделение цен на категории:
SELECT name AS "Услуга", price AS "Цена", 
       IF(price > 3000, 'Дорогая', 'Бюджетная') AS "Категория цены" 
FROM `service`;

-- IF: Проверка доступности сотрудника:
SELECT full_name AS "Сотрудник", 
       IF(status = 'Работает', 'На связи', 'Отсутствует') AS "Доступность" 
FROM `employee`;


Создание одного представления на наиболее популярный запрос (CREATE VIEW) и временное представление – оператор WITH:

-- CREATE VIEW: Постоянный отчет по стоимости работ:
CREATE VIEW `view_request_costs` AS 
SELECT r.request AS "Номер заявки", e.full_name AS "Сотрудник", s.name AS "Услуга", w.hours AS "Часов", s.price AS "Цена за час", (w.hours * s.price) AS "Итоговая стоимость" 
FROM `request` r 
INNER JOIN `worklog` w ON r.request = w.request 
INNER JOIN `service` s ON w.service = s.service 
INNER JOIN `employee` e ON r.employee = e.employee;

-- Вызов представления:
SELECT "Номер заявки", "Сотрудник", "Итоговая стоимость" FROM `view_request_costs`;

-- WITH: Сравнение заявок по двум конкретным проектам:
WITH Petronics_2 AS (
    SELECT r.request AS "Номер заявки", r.problem_description AS "Проблема", p.name AS "Проект" 
    FROM `request` r 
    INNER JOIN `project` p ON r.project = p.project 
    WHERE p.name = 'Petronics 2.0'
), 
Petronics_Mobile AS (
    SELECT r.request AS "Номер заявки", r.problem_description AS "Проблема", p.name AS "Проект" 
    FROM `request` r 
    INNER JOIN `project` p ON r.project = p.project 
    WHERE p.name = 'Petronics Mobile'
) 
SELECT * FROM Petronics_2 
UNION 
SELECT * FROM Petronics_Mobile;


Выполнить анализ и оптимизацию запросов средствами EXPLAIN и созданием индексов (CREATE INDEX):

-- Оптимизация заявок за 15 сентября:
EXPLAIN SELECT request, problem_description, execution_priority FROM `request` WHERE request_date = '2026-09-15';

-- Создаем индекс:
CREATE INDEX `idx_request_date` ON `request`(request_date);

-- Проверка после индекса:
EXPLAIN SELECT request, problem_description, execution_priority FROM `request` WHERE request_date = '2026-09-15';

-- Обычный запрос для Аналитика (с русскими столбцами):
SELECT request AS "Номер заявки", problem_description AS "Описание проблемы", execution_priority AS "Срочность" 
FROM `request` 
WHERE request_date = '2026-09-15';


Использовать механизм транзакций (START TRANSACTION) для своей предметной области:

-- Безопасное закрытие заявки. Если произойдет сбой, статус не изменится.
START TRANSACTION;
UPDATE `request` SET status = 1, user_comment = 'Проблема успешно решена' WHERE request = 2;
SELECT request AS "Номер", status AS "Статус", user_comment AS "Комментарий" FROM `request` WHERE request = 2;
COMMIT;


Один из запросов на выборку записать и выполнить средствами операторов PREPARE, EXECUTE, DEALLOCATE:

-- Безопасный поиск заявки по номеру:
SET @sql_query = 'SELECT request AS "Номер", problem_description AS "Проблема", execution_priority AS "Срочность" FROM `request` WHERE request = ?';
PREPARE stmt FROM @sql_query;
SET @req_id = 3;
EXECUTE stmt USING @req_id;
DEALLOCATE PREPARE stmt;
